<script>
(() => {
  const $ = id => document.getElementById(id);
  const input = $("input");
  const aa = $("aa");
  const pad = $("pad");
  const padOut = $("padOut");
  const env = $("env");
  const envOut = $("envOut");
  const statusInfo = $("statusInfo");
  const diffInfo = $("diffInfo");
  const copyBtn = $("copy");
  const clearBtn = $("clear");
  const guideToggle = $("guideToggle");
  const previewWrap = $("previewWrap");
  const guideLine = $("guideLine");

  const FONTS = {
    "sns-mobile": '-apple-system, BlinkMacSystemFont, "Hiragino Sans", "Hiragino Kaku Gothic ProN", "Yu Gothic UI", sans-serif',
    "sns-pc": '"Hiragino Sans", "Hiragino Kaku Gothic ProN", "Yu Gothic UI", "Meiryo", sans-serif',
    "mono": '"SF Mono", "Menlo", "Monaco", "Consolas", monospace'
  };

  const measure = document.createElement("span");
  measure.style.cssText = 'position:absolute;left:-99999px;top:0;visibility:hidden;white-space:pre;font-size:20px;font-weight:400;line-height:1;letter-spacing:0';
  document.body.appendChild(measure);

  function setMeasureFont(key) {
    const f = FONTS[key] || FONTS["sns-mobile"];
    measure.style.fontFamily = f;
    aa.style.fontFamily = f;
  }

  const px = text => {
    measure.textContent = text || "";
    return measure.getBoundingClientRect().width;
  };

  function graphemes(str) {
    if (window.Intl && Intl.Segmenter) {
      return [...new Intl.Segmenter("ja", { granularity: "grapheme" }).segment(str)].map(x => x.segment);
    }
    return Array.from(str);
  }

  const clean = s => s.replace(/[\r\n]+/g, " ").trim();
  const PLACEHOLDER_RAW = "ラーメン";

  function optimize(chars, forcedPad, fontKey) {
    setMeasureFont(fontKey);
    const padCandidates = forcedPad === 0 ? (chars.length <= 1 ? [1, 2] : [1, 2, 3]) : [forcedPad];
    const innerText = chars.join("　");
    let best = null;

    for (const p of padCandidates) {
      const midText = "＞" + "　".repeat(p) + innerText + "　".repeat(p) + "＜";
      const midWidth = px(midText);

      const renW = Math.max(px("人"), 0.1);
      const topBaseW = px("＿＿");
      const estRen = Math.max(1, Math.round((midWidth - topBaseW) / renW));

      const yW = Math.max(px("^Y"), 0.1);
      const botBaseW = px("￣￣");
      const estY = Math.max(1, Math.round((midWidth - botBaseW) / yW));

      for (let n = Math.max(1, estRen - 3); n <= estRen + 3; n++) {
        const topText = "＿" + "人".repeat(n) + "＿";
        const topWidth = px(topText);

        for (let m = Math.max(1, estY - 3); m <= estY + 3; m++) {
          const variants = ["^Y".repeat(m), "^Y".repeat(m) + "^"];
          for (const pattern of variants) {
            const botText = "￣" + pattern + "￣";
            const botWidth = px(botText);

            const diffTopMid = Math.abs(topWidth - midWidth);
            const diffBotMid = Math.abs(botWidth - midWidth);
            const diffTopBot = Math.abs(topWidth - botWidth);
            const maxDiff = Math.max(diffTopMid, diffBotMid, diffTopBot);

            let score = maxDiff * 3.0 + (diffTopMid + diffBotMid + diffTopBot) * 1.5 + p * 0.4;

            if (!best || score < best.score) {
              best = { topText, midText, botText, maxDiff, p, renCount: n, score };
            }
          }
        }
      }
    }
    return best;
  }

  function render() {
    const raw = clean(input.value);
    const forcedPad = Number(pad.value);
    const selectedEnv = env.value;

    padOut.textContent = forcedPad === 0 ? "自動" : `${forcedPad}`;
    envOut.textContent = env.options[env.selectedIndex].text.split(" ")[0];

    const isPlaceholder = !raw;
    const chars = graphemes(isPlaceholder ? PLACEHOLDER_RAW : raw);
    const res = optimize(chars, forcedPad, selectedEnv);

    if (isPlaceholder) {
      aa.style.opacity = "0.32";
      statusInfo.textContent = `例示中 (未入力) ｜ 文字数: 0字`;
      diffInfo.textContent = `右端ズレ: ~${Math.round(res.maxDiff)}px`;
    } else {
      aa.style.opacity = "1";
      statusInfo.textContent = `文字数: ${chars.length}字 ｜ 上: ${res.renCount}個 ｜ 余白: ${res.p}`;
      const pxDiff = Math.round(res.maxDiff);
      diffInfo.textContent = pxDiff <= 1 ? "右端ズレ: 0px (完全一致)" : `右端ズレ: 約${pxDiff}px`;
      diffInfo.style.color = pxDiff <= 2 ? "var(--success)" : "var(--accent)";
    }

    aa.textContent = `${res.topText}\n${res.midText}\n${res.botText}`;

    requestAnimationFrame(() => {
      const rect = aa.getBoundingClientRect();
      guideLine.style.left = `${rect.width + 14}px`;
    });
  }

  input.addEventListener("input", render);
  pad.addEventListener("input", render);
  env.addEventListener("change", render);

  guideToggle.addEventListener("change", () => {
    previewWrap.classList.toggle("show-guide", guideToggle.checked);
  });

  clearBtn.addEventListener("click", () => {
    input.value = "";
    render();
    input.focus();
  });

  copyBtn.addEventListener("click", async () => {
    let textToCopy = aa.textContent;
    if (!clean(input.value)) {
      const res = optimize(graphemes(PLACEHOLDER_RAW), Number(pad.value), env.value);
      textToCopy = `${res.topText}\n${res.midText}\n${res.botText}`;
    }

    try {
      await navigator.clipboard.writeText(textToCopy);
    } catch (e) {
      const ta = document.createElement("textarea");
      ta.value = textToCopy;
      ta.style.cssText = "position:fixed;opacity:0";
      document.body.appendChild(ta);
      ta.select();
      document.execCommand("copy");
      ta.remove();
    }

    const orig = copyBtn.innerHTML;
    copyBtn.style.background = "var(--success)";
    copyBtn.style.color = "#071a10";
    copyBtn.innerHTML = `✓ コピー完了！`;
    setTimeout(() => {
      copyBtn.style.background = "";
      copyBtn.style.color = "";
      copyBtn.innerHTML = orig;
    }, 1400);
  });

  $("shareX").addEventListener("click", () => {
    window.open(`https://twitter.com/intent/tweet?text=${encodeURIComponent(aa.textContent)}`, "_blank");
  });

  $("shareLine").addEventListener("click", () => {
    window.open(`https://line.me/R/msg/text/?${encodeURIComponent(aa.textContent)}`, "_blank");
  });

  // 自分自身（HTMLファイル）をダウンロード保存するボタン
  $("downloadSelf").addEventListener("click", () => {
    const htmlContent = "<!doctype html>\n" + document.documentElement.outerHTML;
    const blob = new Blob([htmlContent], { type: "text/html;charset=utf-8" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = "index.html";
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  });

  input.value = "";
  render();

  if (document.fonts && document.fonts.ready) {
    document.fonts.ready.then(render);
  }
  window.addEventListener("resize", render);
})();
</script>
